<!-- 直方图：原始数据 → 分bin → Chart.js bar --> function buildHistogram(rawData, binCount) { const min=Math.min(...rawData), max=Math.max(...rawData); const binW=(max-min)/binCount; const bins=Array(binCount).fill(0); rawData.forEach(v=>{ const i=Math.min(Math.floor((v-min)/binW),binCount-1); bins[i]++; }); return { labels: bins.map((_,i)=>(min+i*binW).toFixed(1)+'-'+(min+(i+1)*binW).toFixed(1)), data:bins }; }
